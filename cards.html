function renderCards(employees) { const grid = document.getElementById('cards-grid'); grid.innerHTML = ''; if (!employees || employees.length === 0) { setStatus('表示できる従業員データがありません（0件）。従業員マスタの登録内容を確認してください。', true); return; } const fragment = document.createDocumentFragment(); const qrTargets = []; employees.forEach((emp, idx) => { const empId = String(emp.id || emp.empId || emp['従業員ID'] || '').trim(); const name = String(emp.name || emp['氏名'] || emp.employeeName || '').trim(); const dept = String(emp.dept || emp.deptName || emp['現所属部署'] || emp['部署・所属'] || '').trim(); const status = String(emp.status || emp['在籍ステータス'] || '').trim(); const card = document.createElement('div'); card.className = 'id-card'; const idEl = document.createElement('div'); idEl.className = 'emp-id'; idEl.textContent = `ID: ${empId || '(未設定)'}`; const nameEl = document.createElement('div'); nameEl.className = 'emp-name'; nameEl.textContent = name || '(氏名未設定)'; const deptEl = document.createElement('div'); deptEl.className = 'emp-dept'; deptEl.textContent = dept || '未設定'; const qrEl = document.createElement('div'); qrEl.className = 'qr-box'; qrEl.id = `qr-${idx}`; const subEl = document.createElement('div'); subEl.className = 'emp-sub'; subEl.textContent = status ? `ステータス: ${status}` : 'QR社員証'; card.appendChild(idEl); card.appendChild(nameEl); card.appendChild(deptEl); card.appendChild(qrEl); card.appendChild(subEl); fragment.appendChild(card); if (empId && typeof QRCode !== 'undefined') { qrTargets.push({ el: qrEl, text: empId }); } }); grid.appendChild(fragment); qrTargets.forEach(item => { new QRCode(item.el, { text: item.text, width: 140, height: 140, correctLevel: QRCode.CorrectLevel.M }); }); setStatus(`${employees.length}名分の社員証QRコードを表示しました。`); }
